# The portable core (PLAN.md section 3): no pico-sdk, no hardware (rule 5).
# Built on both platforms from the start, so the RP2350 toolchain keeps it honest.

# §7's built-in font: fonts/cp437-6x8.bin, turned into C in the build tree, so no
# font byte is typed into or committed as C. Configure re-runs when the file changes.
set(PICOVDP_FONT ${CMAKE_CURRENT_SOURCE_DIR}/../fonts/cp437-6x8.bin)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${PICOVDP_FONT})
file(SIZE ${PICOVDP_FONT} PICOVDP_FONT_SIZE)
if(NOT PICOVDP_FONT_SIZE EQUAL 2048)
    message(FATAL_ERROR "${PICOVDP_FONT} is ${PICOVDP_FONT_SIZE} bytes, not 2048")
endif()
file(READ ${PICOVDP_FONT} PICOVDP_FONT_HEX HEX)
string(REGEX REPLACE "([0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f][0-9a-f])" "\\1\n" PICOVDP_FONT_HEX "${PICOVDP_FONT_HEX}")
string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," PICOVDP_FONT_BYTES "${PICOVDP_FONT_HEX}")
file(CONFIGURE OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/font_data.c CONTENT
"// Generated by core/CMakeLists.txt from fonts/cp437-6x8.bin. Do not edit.

#include \"vdp_internal.h\"

const uint8_t vdp_font_cp437[VDP_FONT_BYTES] = {
@PICOVDP_FONT_BYTES@};
" @ONLY)

add_library(picovdp_core STATIC bus.c font.c geometry.c line.c palette.c render.c sprites.c status.c tables.c tiles.c
    ${CMAKE_CURRENT_BINARY_DIR}/font_data.c)
target_include_directories(picovdp_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(picovdp_core PRIVATE picovdp_warnings)

# Inspection and snapshots (vdp_debug.h): the adapter, the unit tests and the
# debug link. Kept out of picovdp_core so a release image need not link it.
add_library(picovdp_core_debug STATIC debug.c)
target_link_libraries(picovdp_core_debug PUBLIC picovdp_core PRIVATE picovdp_warnings)

if(PICOVDP_PLATFORM STREQUAL "host")
    # Linked into the Node-API addon, a shared module.
    set_target_properties(picovdp_core picovdp_core_debug PROPERTIES POSITION_INDEPENDENT_CODE ON)
else()
    target_compile_definitions(picovdp_core PRIVATE PICOVDP_RP2350)
endif()
